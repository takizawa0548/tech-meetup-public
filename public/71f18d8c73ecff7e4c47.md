### 初期化で宣言された型
TypeScriptは宣言された時に右側の式や値から型を推測します。
再宣言時も初期化された型に限定されるため、初期化された型以外で再宣言するとTypeScriptが警告します。
```ts
// x: string | number
let x = Math.random() < 0.5 ? 10 : "hello world!"; 
// x: string | number
// OK
x = 1;
// NG
x = ture;
```

### 型によるfilter絞り込み

```ts
type Fish = { swim: () => void };
type Bird = { fly: () => void };
type Human = { swim?: () => void; fly?: () => void };

function isFish(pet: Fish | Bird): pet is Fish {
  return (pet as Fish).swim !== undefined;
}

const zoo: (Fish | Bird)[] = [
    {swim: () => console.log('swim')}, 
    {swim: () => console.log('swim')}, 
    {fly: () => console.log('fly')}
];
const underWater: Fish[] = zoo.filter(isFish);

// swimだけに絞り込まれるため2
console.log(underWater.length);
```

### 存在すべきではない型（never)
全ての想定されるかたが処理された後に推定される型です。
```ts
type Shape = Circle | Square;
 
function getArea(shape: Shape) {
  switch (shape.kind) {
    case "circle":
      return Math.PI * shape.radius ** 2;
    case "square":
      return shape.sideLength ** 2;
    default:
      // 定義した型が全て対応できていれば警告されません 
      const _exhaustiveCheck: never = shape;
      return _exhaustiveCheck;
  }
}
```
